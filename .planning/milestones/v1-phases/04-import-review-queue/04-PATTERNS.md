### `src/server/imports/store.ts` (service, CRUD in IMMEDIATE tx)

**Analog:** `src/server/discogs/enqueue.ts` `enqueueExplicitSearch` L268-338.

**Discriminated result + IMMEDIATE transaction shape:**
```typescript
export type ExplicitSearchResult =
  | { status: 'ok'; searchId: string } | { status: 'not_found' } | { status: 'not_searchable' } | { status: 'busy' };

export function enqueueExplicitSearch(db: Database.Database, input: ExplicitSearchInput): ExplicitSearchResult {
  const tx = db.transaction((): ExplicitSearchResult => {
    const spine = db.prepare('SELECT state FROM spines WHERE id = @spineId')
      .get({ spineId: input.spineId }) as { state: string } | undefined;
    if (!spine) return { status: 'not_found' };
    if (spine.state !== 'active' && spine.state !== 'orphaned') return { status: 'not_searchable' };
    ... busy check via LEFT JOIN jobs j ... j.status IN ('queued','running') ...
    const job = enqueue({ lane: 'discogs', type, payload: { searchId }, dedupeKey: `search:${searchId}` });
    setSearchJob(db, searchId, job.id);
    return { status: 'ok', searchId };
  });
  return tx.immediate();
}
```

export async function POST(req: Request, { params }: { params: Promise<{ id: string }> }) {
  await connection();
  const contentType = req.headers.get('content-type') ?? '';
  if (!/^application\/json\s*(;|$)/i.test(contentType)) {
    return Response.json({ error: 'unsupported_type', message: 'Send application/json.' }, { status: 415 });
  }
  let raw: unknown;
  try { raw = await req.json(); } catch {
    return Response.json({ error: 'invalid_body', message: 'The request body is not JSON.' }, { status: 400 });

  const parsed = Body.safeParse(raw);

  }
  const { id } = await params;
